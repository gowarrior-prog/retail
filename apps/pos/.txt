  // Quick Edit & Add Panel interactions
  const quickAddBtn = document.getElementById('open-quick-add');

  const cancelDrawerBtn = document.getElementById('cancel-drawer-btn');

  // Form Inputs
  const formTitle = document.getElementById('form-title');
  const formSku = document.getElementById('form-sku');
  const formRetail = document.getElementById('form-retail');
  const formCost = document.getElementById('form-cost');
  const formStock = document.getElementById('form-stock');
  const calcMargin = document.getElementById('calculated-margin');

  function updateMargin() {
    const retail = parseFloat(formRetail.value) || 0;
    const cost = parseFloat(formCost.value) || 0;
    if (retail > 0) {
      const margin = ((retail - cost) / retail) * 100;
      const profit = retail - cost;
      calcMargin.innerText = `${margin.toFixed(1)}% ($${profit.toFixed(2)} profit/unit)`;
    }
  }

  formRetail.addEventListener('input', updateMargin);
  formCost.addEventListener('input', updateMargin);

  // Edit row triggers
  document.querySelectorAll('.edit-row-btn').forEach((btn) => {
    btn.addEventListener('click', (e) => {
      const row = e.target.closest('tr');
      const title = row.querySelector('.group-hover\\:text-primary')?.innerText || 'Apparel Product';
      const sku = row.querySelector('.font-label-badge')?.innerText || 'SKU-000';
      panelTitle.innerText = `Edit: ${sku}`;
      formTitle.value = title;
      formSku.value = sku;
      quickAddPanel.scrollIntoView({ behavior: 'smooth' });
    });
  });

  quickAddBtn.addEventListener('click', () => {
    panelTitle.innerText = "Add New Product";
    formTitle.value = "";
    formSku.value = "APP-" + Math.floor(1000 + Math.random() * 9000);
    formRetail.value = "0.00";
    formCost.value = "0.00";
    formStock.value = "1";
    calcMargin.innerText = "0.0% ($0.00 profit/unit)";
    quickAddPanel.scrollIntoView({ behavior: 'smooth' });
  });

  closeQuickAdd.addEventListener('click', () => {
    quickAddPanel.classList.add('opacity-40');
    setTimeout(() => quickAddPanel.classList.remove('opacity-40'), 200);
  });
  cancelDrawerBtn.addEventListener('click', () => {
    panelTitle.innerText = "Quick Add Product";
  });

  saveProductBtn.addEventListener('click', () => {
    saveProductBtn.innerHTML = `<span class="material-symbols-outlined text-[18px] animate-spin">refresh</span> Saving SKU...`;
    setTimeout(() => {
      saveProductBtn.innerHTML = `<span class="material-symbols-outlined text-[18px]">check</span> SKU Updated!`;
      setTimeout(() => {
        saveProductBtn.innerHTML = `<span class="material-symbols-outlined text-[18px]">check_circle</span> <span>Save Changes (↵ Enter)</span>`;
      }, 1200);
    }, 450);
  });

  // Checkbox bulk actions
  const selectAll = document.getElementById('select-all');
  const productSelects = document.querySelectorAll('.product-select');
  const bulkActionBar = document.getElementById('bulk-action-bar');
  const selectedCount = document.getElementById('selected-count');

  function updateBulkBar() {
    const checked = document.querySelectorAll('.product-select:checked');
    if (checked.length > 0) {
      bulkActionBar.classList.remove('hidden');
      bulkActionBar.classList.add('flex');
      selectedCount.innerText = `${checked.length} selected`;
    } else {
      bulkActionBar.classList.add('hidden');
      bulkActionBar.classList.remove('flex');
    }
  }

  selectAll.addEventListener('change', () => {
    productSelects.forEach(cb => cb.checked = selectAll.checked);
    updateBulkBar();
  });

  productSelects.forEach(cb => {
    cb.addEventListener('change', updateBulkBar);
  });

  // Stock Adjustment Modal logic
  const stockModal = document.getElementById('stock-modal');
  const closeModalBtn = document.getElementById('close-modal-btn');
  const cancelAdjustBtn = document.getElementById('cancel-adjust-btn');
  const confirmAdjustBtn = document.getElementById('confirm-adjust-btn');
  const adjustInput = document.getElementById('adjust-input');
  const decBtn = document.getElementById('decrement-count');
  const incBtn = document.getElementById('increment-count');

      const row = e.target.closest('tr');
      modalProdName.innerText = row.querySelector('.group-hover\\:text-primary')?.innerText || 'Product';
      modalProdSku.innerText = row.querySelector('.font-label-badge')?.innerText || 'SKU';
      stockModal.classList.remove('hidden');
      stockModal.classList.add('flex');
    });
  });

  function closeStockModal() {
    stockModal.classList.add('hidden');
    stockModal.classList.remove('flex');
  }

  closeModalBtn.addEventListener('click', closeStockModal);
  cancelAdjustBtn.addEventListener('click', closeStockModal);

  decBtn.addEventListener('click', () => {
    let val = parseInt(adjustInput.value) || 0;
    adjustInput.value = Math.max(0, val - 1);
  });

  incBtn.addEventListener('click', () => {
    let val = parseInt(adjustInput.value) || 0;
    adjustInput.value = val + 1;

    confirmAdjustBtn.innerText = "Synchronizing...";
    setTimeout(() => {
      confirmAdjustBtn.innerText = "Commit Stock Change";
      closeStockModal();

      e.preventDefault();
      quickAddBtn.click();
    }
    if (e.key === 'Escape') {
      closeStockModal();
    }
